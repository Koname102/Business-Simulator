// ============================================
// FILE: lib/types.ts
// PURPOSE: TypeScript type definitions (MVP only)
// RELATIONS: Used by gameStore.ts, all components
// UPDATED: 2024-11-08 (Initial MVP version)
// ============================================

export type BusinessType = 'fintech' | 'insurance' | 'investment';

export interface Player {
  id: string;
  name: string;
  age: number;
  experience: number;
  level: number;
  createdAt: number;
}

export interface Company {
  id: string;
  name: string;
  type: BusinessType;
  foundedAt: number;
  capital: number;
  balance: number;
  totalRevenue: number;
  totalExpense: number;
  monthlyProfit: number;
  reputation: number;
  customers: number;
  gameTime: number;
  isPaused: boolean;
}

export interface Transaction {
  id: string;
  timestamp: number;
  type: 'income' | 'expense';
  amount: number;
  category: string;
  description: string;
}

export interface GameNotification {
  id: string;
  timestamp: number;
  type: 'info' | 'success' | 'warning' | 'error';
  title: string;
  message: string;
  read: boolean;
}

export interface FintechState {
  loans: Loan[];
  totalDisbursed: number;
  totalRepaid: number;
  defaultRate: number;
  activeLoans: number;
}

export interface Loan {
  id: string;
  borrowerName: string;
  amount: number;
  interestRate: number;
  duration: number;
  status: 'pending' | 'active' | 'paid' | 'defaulted';
  disbursedAt?: number;
  dueDate?: number;
  paidAmount: number;
  creditScore: number;
}

export interface InsuranceState {
  policies: Policy[];
  totalPremiumCollected: number;
  totalClaimsPaid: number;
  claimRatio: number;
  activePolicies: number;
}

export interface Policy {
  id: string;
  holderName: string;
  type: 'life' | 'health' | 'vehicle' | 'property';
  coverageAmount: number;
  premiumMonthly: number;
  status: 'active' | 'expired' | 'claimed';
  startDate: number;
  endDate: number;
}

export interface InvestmentState {
  portfolio: Investment[];
  aum: number;
  totalReturn: number;
  activeInvestments: number;
}

export interface Investment {
  id: string;
  targetName: string;
  amount: number;
  equityStake: number;
  status: 'active' | 'exited' | 'failed';
  investedAt: number;
  currentValuation: number;
  returnMultiple: number;
  sector: string;
}

export interface GameState {
  player: Player;
  company: Company;
  fintech?: FintechState;
  insurance?: InsuranceState;
  investment?: InvestmentState;
  transactions: Transaction[];
  notifications: GameNotification[];
  lastSaved: number;
  version: string;
}

/*
=== TYPE EXPLANATIONS ===

BusinessType
  - Only 3 options: fintech, insurance, investment
  - Selected during company creation
  
Player
  - id: Generated with crypto.randomUUID()
  - name: User input
  - age: Default 25
  - experience: XP points (0 to start)
  - level: Calculated from XP (starts at 1)
  - createdAt: Date.now() timestamp

Company
  - id: Generated with crypto.randomUUID()
  - name: User input
  - type: Selected BusinessType
  - capital: Starting capital (from GAME_CONFIG)
  - balance: Current cash (starts = capital)
  - totalRevenue: All-time income (starts 0)
  - totalExpense: All-time spending (starts 0)
  - monthlyProfit: This month profit (starts 0)
  - reputation: 0-100 score (starts 50)
  - customers: Count (starts 0)
  - gameTime: In-game seconds (starts 0)
  - isPaused: Game pause state (starts false)

Transaction
  - Log of income/expense changes
  - Max 100 recent stored

GameNotification
  - In-game alerts to player
  - Max 50 recent stored

FintechState, InsuranceState, InvestmentState
  - Business-specific data
  - Only ONE active based on company.type
  - Others are undefined

GameState
  - ROOT state object
  - Contains ALL game data
  - Saved to localStorage
  
=== RELATIONS ===
- Imported by: gameStore.ts, all game components
- Used for: Type checking, autocomplete, validation
*/
